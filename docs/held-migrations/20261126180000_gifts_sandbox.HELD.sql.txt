-- Gifts: catalogue, server-priced sending and an append-only gift ledger (sandbox, testers only). Append-only.
--
-- Founder approval is still pending for gift pricing and for any creator share, so this migration invents neither:
--   * the catalogue starts EMPTY; admins add gifts and set their token prices (logged, effective immediately,
--     and every gift sent keeps a snapshot of the price it was sent at);
--   * creator_share_pct is constrained to stay NULL: no part of a gift is credited to anyone. The tokens leave
--     the sender's wallet and the gift is recorded with allocation_status 'pending_policy' until an approved,
--     reviewed migration defines a share. There is no earnings, withdrawal or payout path here;
--   * sending needs the gift sandbox switch (admin, off by default) and BOTH people marked as testers.
--     live_enabled is locked false by a constraint, exactly like the booking and billing switches.

-- Admins add or change a gift. Changes apply to gifts sent afterwards; sent gifts keep their snapshot.
create function public.admin_set_gift(p_key text, p_name text, p_token_price int, p_active boolean)
returns void
language plpgsql
security definer
set search_path = ''
as $$
begin
  perform private.require_role('admin');
  insert into public.gift_catalog (key, name, token_price, active, updated_by, updated_at)
  values (p_key, btrim(p_name), p_token_price, coalesce(p_active, false), (select auth.uid()), now())
  on conflict (key) do update set name = excluded.name, token_price = excluded.token_price,
    active = excluded.active, updated_by = excluded.updated_by, updated_at = now();
  insert into public.gift_price_history (gift_key, token_price, active, changed_by)
  values (p_key, p_token_price, coalesce(p_active, false), (select auth.uid()));
  perform private.log_action('gift_set', 'settings', p_key, null,
    jsonb_build_object('token_price', p_token_price, 'active', coalesce(p_active, false)));
end;
$$;

create function public.admin_set_gift_sandbox(p_enabled boolean)
returns void
language plpgsql
security definer
set search_path = ''
as $$
begin
  perform private.require_role('admin');
  update public.gift_settings set sandbox_enabled = coalesce(p_enabled, false), updated_by = (select auth.uid()),
    updated_at = now() where id = 1;
  perform private.log_action(case when p_enabled then 'gift_sandbox_on' else 'gift_sandbox_off' end,
    'settings', 'gifts', null);
end;
$$;

-- Sends one gift. The same idempotency key returns the first gift and never charges twice.
create function public.gift_send(p_gift text, p_recipient uuid, p_stream uuid, p_key text)
returns bigint
language plpgsql
security definer
set search_path = ''
as $$
declare
  me uuid := private.require_member();
  g public.gift_catalog;
  prior public.gift_transactions;
  st public.live_streams;
  entry bigint;
  gid bigint;
begin
  if char_length(coalesce(p_key, '')) not between 8 and 80 then
    raise exception 'Invalid request key.' using errcode = '22023';
  end if;
  select * into prior from public.gift_transactions where idempotency_key = 'gift:' || p_key;
  if prior.id is not null then
    if prior.sender_id is distinct from me or prior.recipient_id is distinct from p_recipient
       or prior.gift_key is distinct from p_gift or prior.stream_id is distinct from p_stream then
      raise exception 'This request key was already used for a different gift.' using errcode = '23505';
    end if;
    return prior.id;
  end if;
  if not coalesce((select s.sandbox_enabled from public.gift_settings s where s.id = 1), false) then
    raise exception 'Gifts are not available yet.' using errcode = 'P0001';
  end if;
  if p_recipient is null or p_recipient = me then
    raise exception 'Choose who to send the gift to.' using errcode = '22023';
  end if;
  if not exists (select 1 from public.profiles p where p.id = me and p.is_tester and not p.is_demo)
     or not exists (select 1 from public.profiles p where p.id = p_recipient and p.is_tester and not p.is_demo) then
    raise exception 'Gifts are in testing and limited to test accounts.' using errcode = 'P0001';
  end if;
  if not exists (select 1 from public.creators c where c.user_id = p_recipient and c.status = 'approved') then
    raise exception 'Gifts can only be sent to approved creators.' using errcode = 'P0001';
  end if;
  if private.blocked_pair(me, p_recipient) or private.is_suspended(p_recipient) then
    raise exception 'This gift cannot be sent.' using errcode = 'P0001';
  end if;
  if p_stream is not null then
    select * into st from public.live_streams s where s.id = p_stream;
    if st.id is null or st.creator_id <> p_recipient
       or not (private.watch_problems(p_stream, me, null) <@ array['country_unknown']::text[]) then
      raise exception 'Gifts can only be sent during a live stream you can watch.' using errcode = 'P0001';
    end if;
  end if;
  select * into g from public.gift_catalog c where c.key = p_gift and c.active;
  if g.key is null then
    raise exception 'That gift is not available.' using errcode = 'P0001';
  end if;
  entry := private.ledger_post(me, -g.token_price, 'gift', 'gift', p_gift, 'gift:' || p_key);
  insert into public.gift_transactions (sender_id, recipient_id, gift_key, gift_name, tokens, stream_id, ledger_entry,
                                        idempotency_key)
  values (me, p_recipient, g.key, g.name, g.token_price, p_stream, entry, 'gift:' || p_key)
  returning id into gid;
  return gid;
end;
$$;

-- Both sides of a member's gift history (newest first).
create function public.my_gift_history(p_limit int default 50)
returns table (id bigint, direction text, other_id uuid, gift_name text, tokens int, stream_id uuid,
               allocation_status text, created_at timestamptz)
language sql
stable
security definer
set search_path = ''
as $$
  select t.id, case when t.sender_id = (select auth.uid()) then 'sent' else 'received' end,
         case when t.sender_id = (select auth.uid()) then t.recipient_id else t.sender_id end,
         t.gift_name, t.tokens, t.stream_id, t.allocation_status, t.created_at
  from public.gift_transactions t
  where (select auth.uid()) in (t.sender_id, t.recipient_id)
  order by t.created_at desc
  limit least(greatest(coalesce(p_limit, 50), 1), 200);
$$;

-- Admin reconciliation: every gift's ledger debit matches its snapshot price exactly once.
create function public.admin_gift_reconciliation()
returns table (gifts bigint, tokens bigint, ledger_tokens bigint, balanced boolean)
language plpgsql
stable
security definer
set search_path = ''
as $$
begin
  perform private.require_role('moderator');
  return query
  select count(*)::bigint, coalesce(sum(t.tokens), 0)::bigint,
         coalesce(-(select sum(l.delta) from public.token_ledger l where l.reason = 'gift'), 0)::bigint,
         coalesce(sum(t.tokens), 0) = coalesce(-(select sum(l.delta) from public.token_ledger l where l.reason = 'gift'), 0)
           and not exists (select 1 from public.gift_transactions t2 join public.token_ledger l2 on l2.id = t2.ledger_entry
                           where l2.delta <> -t2.tokens or l2.user_id is distinct from t2.sender_id)
  from public.gift_transactions t;
end;
$$;
